{{- /* URL of the image most likely to be the Largest Contentful Paint element: the page's cover, or on the home page the cover of the first card. Returns an empty string when there is nothing to preload. */ -}} {{- $url := "" -}} {{- $page := . -}} {{- with .Params.image -}} {{- $url = partial "media-url.html" (dict "src" (.path | default .) "subpath" $page.Params.media_subpath) -}} {{- end -}} {{- if and (not $url) (eq .Kind "home") -}} {{- $all := where .Site.RegularPages "Type" "posts" -}} {{- $pinned := sort (uniq (append (where $all "Params.pin" true) (where $all "Params.pin" "true"))) "Date" "desc" -}} {{- $candidate := "" -}} {{- with $pinned -}} {{- $candidate = index . 0 -}} {{- else -}} {{- range $all -}} {{- if and (not $candidate) (not .Params.hidden) -}}{{ $candidate = . }}{{- end -}} {{- end -}} {{- end -}} {{- with $candidate -}} {{- with .Params.image -}} {{- $url = partial "media-url.html" (dict "src" (.path | default .) "subpath" $candidate.Params.media_subpath) -}} {{- end -}} {{- end -}} {{- end -}} {{- return $url -}}

{{- /* Ports _includes/post-summary.html. Context: dict "post" (page) "full_text" (bool) "max_length" (int). Returns the summary as a string (D-03: Jekyll truncate semantics). */ -}} {{- $post := .post -}} {{- $full := .full_text | default false -}} {{- if and $post.Description (not $full) -}} {{- $post.Description -}} {{- else -}} {{- $content := .content | default $post.Content -}} {{- if strings.Contains $content `
` -}}
    {{- $content = replace $content `
` `` -}}
  {{- end -}}
  {{- $text := "" -}}
  {{- if .keep_entities -}}
    {{- /* Jekyll's strip_html does not decode character references, and the
           search index is compared with it byte for byte. */ -}}
    {{- $text = $content -}}
    {{- $text = replaceRE `(?s)` "" $text -}}
    {{- $text = replaceRE `(?s)` "" $text -}}
    {{- $text = replaceRE `(?s)` "" $text -}}
    {{- $text = replaceRE `` " " $text -}}
    {{- $text = replaceRE `` " " $text -}}
    {{- $text = replaceRE `<[^>]+>` "" $text -}}
    {{- /* kramdown's markdownify unescapes backslash-escaped punctuation before
           whitespace is normalised: a LaTeX hard break (`\\` at the end of a
           line) becomes one backslash joined to the next line, and the
           delimiters become literal brackets. */ -}}
    {{- $text = replaceRE `\\\\\s*\n\s*` `\` $text -}}
    {{- $text = replaceRE `\\\\` `\` $text -}}
    {{- $text = replaceRE `\\\[` `[` $text -}}
    {{- $text = replaceRE `\\\]` `]` $text -}}
    {{- /* Whitespace is collapsed: the index feeds a tokenizer, and Goldmark's
           block formatting would otherwise leave runs of blanks. */ -}}
    {{- $text = replaceRE `\s+` " " $text | strings.TrimSpace -}}
  {{- else -}}
    {{- $text = partial "plain-text.html" $content -}}
  {{- end -}}
  {{- if not $full -}}
    {{- $max := .max_length | default 200 -}}
    {{- if gt (strings.RuneCount $text) $max -}}
      {{- $text = printf "%s..." (substr $text 0 (sub $max 3)) -}}
    {{- end -}}
  {{- end -}}
  {{- if .keep_entities -}}
    {{- /* The search index is JSON: character references must survive as
           written, not be HTML-escaped a second time. */ -}}
    {{- $text | safeHTML -}}
  {{- else -}}
    {{- $text -}}
  {{- end -}}
{{- end -}}

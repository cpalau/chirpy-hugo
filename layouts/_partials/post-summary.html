{{- /* Ports _includes/post-summary.html. Context: dict "post" (page) "full_text" (bool) "max_length" (int). Returns the summary as a string (D-03: Jekyll truncate semantics). */ -}} {{- $post := .post -}} {{- $full := .full_text | default false -}} {{- if and $post.Description (not $full) -}} {{- $post.Description -}} {{- else -}} {{- $content := $post.Content -}} {{- if strings.Contains $content `
` -}}
    {{- $content = replace $content `
` `` -}}
  {{- end -}}
  {{- $text := partial "plain-text.html" $content -}}
  {{- if not $full -}}
    {{- $max := .max_length | default 200 -}}
    {{- if gt (strings.RuneCount $text) $max -}}
      {{- $text = printf "%s..." (substr $text 0 (sub $max 3)) -}}
    {{- end -}}
  {{- end -}}
  {{- $text -}}
{{- end -}}

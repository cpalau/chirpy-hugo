{{- /* Jekyll's `xml_escape`: `String#encode(xml: :attr)` without the surrounding quotes. Returns a string. */ -}} {{- $s := . | string -}} {{- $s = replace $s "&" "&" -}} {{- $s = replace $s "<" "<" -}} {{- $s = replace $s ">" ">" -}} {{- $s = replace $s "\"" """ -}} {{- $s = replace $s "'" "'" -}} {{- return $s -}}

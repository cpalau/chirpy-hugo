{{- /* Responsive cover: AVIF + WebP + JPEG fallback. Covers are page bundle resources (`cover.jpg`), so they are resized at build time and served with `srcset`. The goal is the LCP: a card cover is displayed at ~380 px and used to be served at 1200 px (150 KB). AVIF at 1200 px brings it down to ~40-70 KB. Context: dict "page" "src" "alt" "sizes" "priority" "mode" "class". - mode "preview" (card) wraps it in
- mode "popup" (post cover) wraps it in - priority=true => loading="eager" fetchpriority="high" (LCP image) When the src is not a bundle resource (e.g. /assets/img/...), it delegates to image.html and keeps the previous behaviour. */ -}} {{- $page := .page -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $sizes := .sizes | default "100vw" -}} {{- $priority := .priority | default false -}} {{- $mode := .mode | default "preview" -}} {{- $extra := .class | default "" -}} {{- $lqip := .lqip | default "" -}} {{- $width := .width | default "" -}} {{- $height := .height | default "" -}} {{- $res := "" -}} {{- with $src -}} {{- $rel := strings.TrimPrefix "/" . -}} {{- with $page.Resources.GetMatch $rel -}} {{- if in (slice "jpeg" "png" "webp") (lower .MediaType.SubType) -}}{{ $res = . }}{{- end -}} {{- end -}} {{- if not $res -}} {{- with $page.Resources.GetMatch (path.Base $rel) -}} {{- if in (slice "jpeg" "png" "webp") (lower .MediaType.SubType) -}}{{ $res = . }}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if $lqip -}} {{- partial "image.html" (dict "src" $src "alt" $alt "class" (printf "preview-img %s" $extra) "lqip" $lqip "width" $width "height" $height "page" $page "mode" $mode) -}} {{- else if $res -}} {{- $w := $res.Width -}} {{- $widths := slice -}} {{- range (slice 480 800 1200) -}} {{- if le . $w -}}{{ $widths = $widths | append . }}{{- end -}} {{- end -}} {{- if eq (len $widths) 0 -}}{{ $widths = slice $w }}{{- end -}} {{- if lt (index $widths (sub (len $widths) 1)) $w -}}{{ $widths = $widths | append $w }}{{- end -}} {{- $avif := slice -}} {{- $webp := slice -}} {{- range $widths -}} {{- $a := $res.Resize (printf "%dx avif q45" .) -}} {{- $p := $res.Resize (printf "%dx webp q78" .) -}} {{- $avif = $avif | append (printf "%s %dw" $a.RelPermalink .) -}} {{- $webp = $webp | append (printf "%s %dw" $p.RelPermalink .) -}} {{- end -}} {{- $fallback := $res.Resize (printf "%dx q82" (index $widths (sub (len $widths) 1))) -}} {{- $attrs := cond $priority `loading="eager" fetchpriority="high" decoding="async"` `loading="lazy" decoding="async" fetchpriority="low"` -}} {{- $pic := printf `` (delimit $avif ", ") $sizes (delimit $webp ", ") $sizes $fallback.RelPermalink $res.Width $res.Height $alt $attrs -}} {{- if eq $mode "popup" -}} {{- printf `%s` $res.RelPermalink $extra $pic | safeHTML -}} {{- else -}} {{- printf `

{{- /* Preload of the LCP image. It must match EXACTLY the emitted by cover-picture.html: same srcset and same `sizes`. If they differ, the browser downloads the image twice. Separators are emitted as ` ` (like the viewport in head.html): Hugo's minifier turns the spaces of an unquoted attribute into `%20`, which would break the srcset. With the entity, the browser sees real spaces. Context: dict "page" "src" "sizes". */ -}} {{- $page := .page -}} {{- $sizes := .sizes | default "100vw" -}} {{- $res := "" -}} {{- with .src -}} {{- $rel := strings.TrimPrefix "/" . -}} {{- with $page.Resources.GetMatch $rel -}} {{- if in (slice "jpeg" "png" "webp") (lower .MediaType.SubType) -}}{{ $res = . }}{{- end -}} {{- end -}} {{- if not $res -}} {{- with $page.Resources.GetMatch (path.Base $rel) -}} {{- if in (slice "jpeg" "png" "webp") (lower .MediaType.SubType) -}}{{ $res = . }}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if $res -}} {{- $w := $res.Width -}} {{- $widths := slice -}} {{- range (slice 480 800 1200) -}} {{- if le . $w -}}{{ $widths = $widths | append . }}{{- end -}} {{- end -}} {{- if eq (len $widths) 0 -}}{{ $widths = slice $w }}{{- end -}} {{- if lt (index $widths (sub (len $widths) 1)) $w -}}{{ $widths = $widths | append $w }}{{- end -}} {{- $avif := slice -}} {{- range $widths -}} {{- $a := $res.Resize (printf "%dx avif q45" .) -}} {{- $avif = $avif | append (printf "%s %dw" $a.RelPermalink .) -}} {{- end -}} {{- printf `` (delimit $avif ", ") $sizes | safeHTML -}} {{- else -}} {{- with .src -}} {{- end -}} {{- end -}}

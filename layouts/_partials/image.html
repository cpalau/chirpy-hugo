{{- /* Final image markup, shared by render-image.html and the `img` shortcode. Reproduces refactor-content's R4/R5: classes move to the wrapper, width and height stay on the , loading="lazy" unless there is a LQIP. Context: dict src alt class width height lqip page mode ("popup" | "preview"). */ -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $class := .class | default "" -}} {{- $lqip := .lqip | default "" -}} {{- $subpath := .page.Params.media_subpath -}} {{- $url := partial "media-url.html" (dict "src" $src "subpath" $subpath) -}} {{- $wrapperClasses := slice -}} {{- $imgClasses := slice -}} {{- $dims := slice -}} {{- range $c := split $class " " -}} {{- if $c -}} {{- if or (in (slice "preview-img" "normal" "left" "right" "light" "dark") $c) -}} {{- $wrapperClasses = $wrapperClasses | append $c -}} {{- else if and (hasPrefix $c "w-") (ne $c "w-") -}} {{- $wrapperClasses = $wrapperClasses | append $c -}} {{- else -}} {{- $imgClasses = $imgClasses | append $c -}} {{- end -}} {{- end -}} {{- end -}} {{- with .width }}{{ $dims = $dims | append (printf `width="%v"` .) }}{{ end -}} {{- with .height }}{{ $dims = $dims | append (printf `height="%v"` .) }}{{ end -}} {{- $img := "" -}} {{- if $lqip -}} {{- $img = printf `" $img -}} {{- $wrap := printf "popup img-link %s" (delimit $wrapperClasses " ") -}} {{- if eq (.mode | default "popup") "preview" -}} {{- printf `
%s
` (delimit $wrapperClasses " ") $img | safeHTML -}} {{- else -}} {{- printf `%s` $url $wrap $img | safeHTML -}} {{- end -}}

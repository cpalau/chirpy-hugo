{{- /* Final image markup, shared by render-image.html and the `img` shortcode. Reproduces refactor-content's R4/R5: classes move to the wrapper, width and height stay on the , loading="lazy" unless there is a LQIP. Context: dict src alt class width height lqip page mode ("popup" | "preview"). If `src` is a raster resource of the page bundle, the image is served as a with AVIF/WebP srcset (480/800/1200w) and the original as the fallback, so a 1280px screenshot is not sent to a 380px slot. Images that are not bundle resources keep the upstream single-file path. */ -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $class := .class | default "" -}} {{- $lqip := .lqip | default "" -}} {{- $subpath := .page.Params.media_subpath -}} {{- $url := partial "media-url.html" (dict "src" $src "subpath" $subpath) -}} {{- $wrapperClasses := slice -}} {{- $imgClasses := slice -}} {{- $dims := slice -}} {{- range $c := split $class " " -}} {{- if $c -}} {{- if or (in (slice "preview-img" "normal" "left" "right" "light" "dark") $c) -}} {{- $wrapperClasses = $wrapperClasses | append $c -}} {{- else if and (hasPrefix $c "w-") (ne $c "w-") -}} {{- $wrapperClasses = $wrapperClasses | append $c -}} {{- else -}} {{- $imgClasses = $imgClasses | append $c -}} {{- end -}} {{- end -}} {{- end -}} {{- $res := "" -}} {{- if and (not $lqip) $src -}} {{- with .page.Resources.GetMatch (path.Base $src) -}} {{- if in (slice "jpeg" "png" "webp") (lower .MediaType.SubType) -}}{{ $res = . }}{{- end -}} {{- end -}} {{- end -}} {{- $img := "" -}} {{- if $res -}} {{- $w := $res.Width -}} {{- $widths := slice -}} {{- range (slice 480 800 1200) -}} {{- if le . $w -}}{{ $widths = $widths | append . }}{{- end -}} {{- end -}} {{- if eq (len $widths) 0 -}}{{ $widths = slice $w }}{{- end -}} {{- if lt (index $widths (sub (len $widths) 1)) $w -}}{{ $widths = $widths | append $w }}{{- end -}} {{- $avif := slice -}} {{- $webp := slice -}} {{- range $widths -}} {{- $a := $res.Resize (printf "%dx avif q45" .) -}} {{- $p := $res.Resize (printf "%dx webp q78" .) -}} {{- $avif = $avif | append (printf "%s %dw" $a.RelPermalink .) -}} {{- $webp = $webp | append (printf "%s %dw" $p.RelPermalink .) -}} {{- end -}} {{- $sizes := "(min-width: 1200px) 760px, (min-width: 768px) 90vw, 100vw" -}} {{- $dims = slice (printf `width="%d"` $res.Width) (printf `height="%d"` $res.Height) -}} {{- $img = printf `` $img -}} {{- else if not $lqip -}} {{- $img = printf `%s loading="lazy"` $img -}} {{- end -}} {{- if not $res }}{{ $img = printf "%s>" $img }}{{ end -}} {{- $href := $url -}} {{- if $res }}{{ $href = $res.RelPermalink }}{{ end -}} {{- $wrap := printf "popup img-link %s" (delimit $wrapperClasses " ") -}} {{- if eq (.mode | default "popup") "preview" -}} {{- printf `
%s
` (delimit $wrapperClasses " ") $img | safeHTML -}} {{- else -}} {{- printf `%s` $href $wrap $img | safeHTML -}} {{- end -}}

{{- /* Ports _layouts/home.html, including the pinned-first pagination algorithm (G-10). */ -}} {{- define "main" -}} {{- $all := partial "posts-visible.html" . -}} {{- $pag := .Paginate $all -}} {{- $perPage := $pag.PagerSize -}} {{- $allPinned := sort (uniq (append (where $all "Params.pin" true) (where $all "Params.pin" "true"))) "Date" "desc" -}} {{- $allNormal := slice -}} {{- range $all -}} {{- $pin := printf "%v" .Params.pin -}} {{- if and (ne $pin "true") (not .Params.hidden) -}} {{- $allNormal = $allNormal | append . -}} {{- end -}} {{- end -}} {{- $visibleStart := mul (sub $pag.PageNumber 1) $perPage -}} {{- $visibleEnd := add $visibleStart $perPage -}} {{- $posts := slice -}} {{- $pinnedSize := 0 -}} {{- if gt (len $allPinned) $visibleStart -}} {{- if gt (len $allPinned) $visibleEnd -}} {{- $pinnedSize = $perPage -}} {{- else -}} {{- $pinnedSize = sub (len $allPinned) $visibleStart -}} {{- end -}} {{- $posts = after $visibleStart $allPinned | first $pinnedSize -}} {{- end -}} {{- $normalSize := sub (len $pag.Pages) $pinnedSize -}} {{- if gt $normalSize 0 -}} {{- $normalStart := 0 -}} {{- if eq $pinnedSize 0 -}} {{- $normalStart = sub $visibleStart (len $allPinned) -}} {{- end -}} {{- $posts = $posts | append (after $normalStart $allNormal | first $normalSize) -}} {{- end -}} {{- /* The cards use h2 (several per page); the listing needs its own h1 for the heading hierarchy and the SEO audit. */ -}}

{{- /* Rouge-compatible code block (D-02, G-02). Chroma produces the token stream; this hook rebuilds Rouge's DOM and prepends the code header that upstream generates in refactor-content. */ -}} {{- $lang := .Type | default "text" -}} {{- /* kramdown has no mermaid lexer: it emits a plain pre/code block, and the Mermaid script looks for exactly that. */ -}} {{- if eq $lang "mermaid" -}} 
{{ .Inner }}
{{- else -}} {{- $res := transform.HighlightCodeBlock . (dict "noClasses" false "lineNos" false) -}} {{- $inner := $res.Inner -}} {{- $inner = replace $inner `` "" -}} {{- $inner = replace $inner `` "" -}} {{- /* Rouge ends the block with a newline *inside* the last token span; a newline after would add an empty line to the 
 and 22px to the page. */ -}}
{{- if hasSuffix $inner "" -}}
  {{- $before := strings.TrimSuffix "" $inner -}}
  {{- if not (hasSuffix $before "\n") -}}
    {{- $inner = printf "%s\n" $before -}}
  {{- end -}}
{{- else if not (hasSuffix $inner "\n") -}}
  {{- $inner = printf "%s\n" $inner -}}
{{- end -}}
{{- $code := strings.TrimSuffix "\n" (printf "%s" .Inner) -}}
{{- $lines := cond (eq $code "") 0 (len (split $code "\n")) -}}
{{- $nolineno := false -}}
{{- /* Only the explicit attribute adds the class: the SCSS hides the gutter of
       console/terminal/plaintext by language, not by class. */ -}}
{{- /* `{.nolineno}` arrives as a CSS class, not as an attribute. */ -}}
{{- if in (split (.Attributes.class | default "") " ") "nolineno" }}{{ $nolineno = true }}{{ end -}}
{{- $file := .Attributes.file -}}
{{- $label := $file -}}
{{- $icon := "far fa-file-code fa-fw" -}}
{{- if not $file -}}
  {{- $label = partial "code-language.html" $lang -}}
  {{- $icon = "fas fa-code fa-fw small" -}}
{{- end -}}

{{ end }}
